### 引言

在之前的文章中我们介绍了C++20协程的基本原理。但协程本身只是一个暂停/恢复的机制——它不会自己调度，也不会自己与IO事件关联。要让协程真正发挥威力，需要一个**协程调度器（Scheduler）**来驱动它们。

这篇文章从零实现一个mini异步运行时，支持：多线程工作窃取调度、Timer定时器、基于epoll的异步IO。完成后你将理解liburing、asio等框架底层的核心设计思路。

### 1. 协程调度器的职责

```

│                                                                  │

### 2. 基础Task类型

#### 2.1 协程的Promise和Task

```cpp
#include <coroutine>
#include <optional>
#include <exception>

// Task<T>: 代表一个可co_await的异步任务
template<typename T = void>
class Task {
public:
    struct promise_type {
        std::optional<T> value_;
        std::exception_ptr exception_;
        std::coroutine_handle<> continuation_;  // 谁在等我完成

        Task get_return_object() {
            return Task{std::coroutine_handle<promise_type>::from_promise(*this)};
        }

        std::suspend_always initial_suspend() noexcept { return {}; }
        // 初始挂起：创建后不立即执行，由调度器决定何时开始

        auto final_suspend() noexcept {
            // 完成时恢复等待者
            struct FinalAwaiter {
                bool await_ready() noexcept { return false; }
                std::coroutine_handle<> await_suspend(
                    std::coroutine_handle<promise_type> h) noexcept {
                    if (h.promise().continuation_)
                        return h.promise().continuation_;  // 恢复等待者
                    return std::noop_coroutine();          // 没人等，结束
                }
                void await_resume() noexcept {}
            };
            return FinalAwaiter{};
        }

        void return_value(T val) { value_ = std::move(val); }
        void unhandled_exception() { exception_ = std::current_exception(); }
    };

    // co_await Task时的行为
    bool await_ready() const noexcept { return handle_.done(); }

    std::coroutine_handle<> await_suspend(std::coroutine_handle<> caller) {
        handle_.promise().continuation_ = caller;  // 记录谁在等我
        return handle_;  // 对称转移：开始执行本Task
    }

    T await_resume() {
        if (handle_.promise().exception_)
            std::rethrow_exception(handle_.promise().exception_);
        return std::move(*handle_.promise().value_);
    }

    std::coroutine_handle<promise_type> handle() const { return handle_; }

    ~Task() {
        if (handle_) handle_.destroy();
    }

    // Move only
    Task(Task&& other) : handle_(std::exchange(other.handle_, nullptr)) {}
    Task& operator=(Task&&) = delete;

private:
    explicit Task(std::coroutine_handle<promise_type> h) : handle_(h) {}
    std::coroutine_handle<promise_type> handle_;
};
```

### 3. 调度器实现

#### 3.1 单线程调度器（基础版）

```cpp
#include <queue>
#include <coroutine>

class Scheduler {
public:
    // 提交一个协程到调度器
    void spawn(Task<void> task) {
        ready_queue_.push(task.handle());
    }

    // 运行直到所有协程完成
    void run() {
        while (!ready_queue_.empty()) {
            auto handle = ready_queue_.front();
            ready_queue_.pop();

            if (!handle.done()) {
                handle.resume();  // 恢复执行直到下次co_await
            }
        }
    }

    // 让当前协程让出执行权（yield）
    auto yield() {
        struct YieldAwaiter {
            Scheduler* scheduler;
            bool await_ready() noexcept { return false; }
            void await_suspend(std::coroutine_handle<> h) {
                scheduler->ready_queue_.push(h);  // 重新入队
            }
            void await_resume() noexcept {}
        };
        return YieldAwaiter{this};
    }

private:
    std::queue<std::coroutine_handle<>> ready_queue_;
};
```

#### 3.2 多线程调度器（工作窃取）

```cpp
#include <thread>
#include <vector>
#include <deque>
#include <mutex>
#include <condition_variable>
#include <atomic>

class MultiThreadScheduler {
public:
    MultiThreadScheduler(int num_threads = std::thread::hardware_concurrency())
        : num_threads_(num_threads), local_queues_(num_threads) {
        for (int i = 0; i < num_threads; ++i) {
            workers_.emplace_back([this, i]() { workerLoop(i); });
        }
    }

    ~MultiThreadScheduler() {
        shutdown();
    }

    // 提交任务到全局队列
    void spawn(Task<void> task) {
        {
            std::lock_guard<std::mutex> lock(global_mutex_);
            global_queue_.push_back(task.handle());
        }
        cv_.notify_one();
        task_count_.fetch_add(1);
    }

    // 等待所有任务完成
    void waitAll() {
        std::unique_lock<std::mutex> lock(done_mutex_);
        done_cv_.wait(lock, [this]() { return task_count_.load() == 0; });
    }

    // 获取当前worker的schedule awaiter
    auto schedule() {
        struct ScheduleAwaiter {
            MultiThreadScheduler* scheduler;
            bool await_ready() noexcept { return false; }
            void await_suspend(std::coroutine_handle<> h) {
                // 将协程提交到当前worker的本地队列
                int worker_id = scheduler->currentWorkerId();
                if (worker_id >= 0) {
                    scheduler->localEnqueue(worker_id, h);
                } else {
                    scheduler->globalEnqueue(h);
                }
            }
            void await_resume() noexcept {}
        };
        return ScheduleAwaiter{this};
    }

private:
    int num_threads_;
    std::vector<std::thread> workers_;
    std::atomic<bool> running_{true};
    std::atomic<int> task_count_{0};

    // 全局队列（跨线程提交用）
    std::deque<std::coroutine_handle<>> global_queue_;
    std::mutex global_mutex_;
    std::condition_variable cv_;

    // 每个worker的本地队列（减少锁竞争）
    struct alignas(64) LocalQueue {
        std::deque<std::coroutine_handle<>> queue;
        std::mutex mutex;
    };
    std::vector<LocalQueue> local_queues_;

    // 完成通知
    std::mutex done_mutex_;
    std::condition_variable done_cv_;

    static thread_local int t_worker_id_;

    void workerLoop(int id) {
        t_worker_id_ = id;

        while (running_) {
            auto handle = getWork(id);
            if (!handle) continue;

            handle.resume();

            if (handle.done()) {
                if (task_count_.fetch_sub(1) == 1) {
                    done_cv_.notify_all();
                }
            }
        }
    }

    std::coroutine_handle<> getWork(int id) {
        // 优先级1：本地队列
        {
            std::lock_guard<std::mutex> lock(local_queues_[id].mutex);
            if (!local_queues_[id].queue.empty()) {
                auto h = local_queues_[id].queue.front();
                local_queues_[id].queue.pop_front();
                return h;
            }
        }

        // 优先级2：全局队列
        {
            std::lock_guard<std::mutex> lock(global_mutex_);
            if (!global_queue_.empty()) {
                auto h = global_queue_.front();
                global_queue_.pop_front();
                return h;
            }
        }

        // 优先级3：从其他worker偷取（work stealing）
        for (int i = 1; i < num_threads_; ++i) {
            int victim = (id + i) % num_threads_;
            std::lock_guard<std::mutex> lock(local_queues_[victim].mutex);
            if (!local_queues_[victim].queue.empty()) {
                // 从尾部偷（LIFO偷取，FIFO本地执行）
                auto h = local_queues_[victim].queue.back();
                local_queues_[victim].queue.pop_back();
                return h;
            }
        }

        // 无工作可做，等待
        std::unique_lock<std::mutex> lock(global_mutex_);
        cv_.wait_for(lock, std::chrono::milliseconds(1));
        return nullptr;
    }

    void localEnqueue(int id, std::coroutine_handle<> h) {
        std::lock_guard<std::mutex> lock(local_queues_[id].mutex);
        local_queues_[id].queue.push_back(h);
    }

    void globalEnqueue(std::coroutine_handle<> h) {
        std::lock_guard<std::mutex> lock(global_mutex_);
        global_queue_.push_back(h);
        cv_.notify_one();
    }

    int currentWorkerId() { return t_worker_id_; }

    void shutdown() {
        running_ = false;
        cv_.notify_all();
        for (auto& w : workers_) {
            if (w.joinable()) w.join();
        }
    }
};

thread_local int MultiThreadScheduler::t_worker_id_ = -1;
```

### 4. Timer支持

#### 4.1 实现sleep_for

```cpp
#include <chrono>
#include <set>

class TimerManager {
public:
    using Clock = std::chrono::steady_clock;
    using TimePoint = Clock::time_point;

    struct TimerEntry {
        TimePoint deadline;
        std::coroutine_handle<> handle;
        bool operator<(const TimerEntry& o) const { return deadline < o.deadline; }
    };

    // 注册一个定时器
    void addTimer(TimePoint deadline, std::coroutine_handle<> handle) {
        std::lock_guard<std::mutex> lock(mutex_);
        timers_.insert({deadline, handle});
    }

    // 获取所有已到期的协程
    std::vector<std::coroutine_handle<>> pollExpired() {
        std::vector<std::coroutine_handle<>> expired;
        auto now = Clock::now();

        std::lock_guard<std::mutex> lock(mutex_);
        while (!timers_.empty() && timers_.begin()->deadline <= now) {
            expired.push_back(timers_.begin()->handle);
            timers_.erase(timers_.begin());
        }
        return expired;
    }

    // 距下一个定时器的等待时间
    std::chrono::milliseconds nextTimeout() const {
        std::lock_guard<std::mutex> lock(mutex_);
        if (timers_.empty()) return std::chrono::milliseconds(100);
        auto remaining = timers_.begin()->deadline - Clock::now();
        return std::max(std::chrono::milliseconds(0),
                       std::chrono::duration_cast<std::chrono::milliseconds>(remaining));
    }

private:
    std::set<TimerEntry> timers_;
    mutable std::mutex mutex_;
};

// co_await sleep_for(100ms) 的实现
auto sleep_for(std::chrono::milliseconds duration) {
    struct SleepAwaiter {
        TimerManager* timer_mgr;
        std::chrono::milliseconds duration;

        bool await_ready() noexcept { return duration.count() <= 0; }

        void await_suspend(std::coroutine_handle<> h) {
            auto deadline = TimerManager::Clock::now() + duration;
            timer_mgr->addTimer(deadline, h);
        }

        void await_resume() noexcept {}
    };
    return SleepAwaiter{&global_timer_manager, duration};
}

### 5. 异步IO（epoll集成）

#### 5.1 AsyncSocket

```cpp
#include <sys/epoll.h>
#include <unistd.h>
#include <fcntl.h>

class IOLoop {
public:
    IOLoop() {
        epoll_fd_ = epoll_create1(0);
    }

    // 等待fd可读
    auto waitReadable(int fd) {
        struct ReadAwaiter {
            IOLoop* loop;
            int fd;

            bool await_ready() noexcept { return false; }
            void await_suspend(std::coroutine_handle<> h) {
                loop->registerEvent(fd, EPOLLIN | EPOLLONESHOT, h);
            }
            void await_resume() noexcept {}
        };
        return ReadAwaiter{this, fd};
    }

    // 等待fd可写
    auto waitWritable(int fd) {
        struct WriteAwaiter {
            IOLoop* loop;
            int fd;

            bool await_ready() noexcept { return false; }
            void await_suspend(std::coroutine_handle<> h) {
                loop->registerEvent(fd, EPOLLOUT | EPOLLONESHOT, h);
            }
            void await_resume() noexcept {}
        };
        return WriteAwaiter{this, fd};
    }

    // 轮询IO事件，返回就绪的协程
    std::vector<std::coroutine_handle<>> poll(int timeout_ms) {
        epoll_event events[64];
        int n = epoll_wait(epoll_fd_, events, 64, timeout_ms);

        std::vector<std::coroutine_handle<>> ready;
        for (int i = 0; i < n; ++i) {
            auto h = std::coroutine_handle<>::from_address(events[i].data.ptr);
            ready.push_back(h);
        }
        return ready;
    }

private:
    int epoll_fd_;

    void registerEvent(int fd, uint32_t events, std::coroutine_handle<> h) {
        epoll_event ev{};
        ev.events = events;
        ev.data.ptr = h.address();  // 将协程handle存入epoll_event
        epoll_ctl(epoll_fd_, EPOLL_CTL_MOD, fd, &ev);
        // 注意：fd需要先EPOLL_CTL_ADD过，这里简化了
    }
};

// 异步读取
Task<ssize_t> async_read(IOLoop& io, int fd, void* buf, size_t len) {
    co_await io.waitReadable(fd);
    ssize_t n = ::read(fd, buf, len);
    co_return n;
}

// 异步写入
Task<ssize_t> async_write(IOLoop& io, int fd, const void* buf, size_t len) {
    co_await io.waitWritable(fd);
    ssize_t n = ::write(fd, buf, len);
    co_return n;
}

### 6. 完整的事件循环

#### 6.1 整合Timer + IO + 协程调度

```cpp
class EventLoop {
public:
    EventLoop() : io_loop_(), timer_mgr_() {}

    void spawn(Task<void> task) {
        ready_queue_.push_back(task.handle());
    }

    void run() {
        while (!ready_queue_.empty() || hasTimers() || hasIOWaiters()) {
            // 1. 执行所有就绪协程
            while (!ready_queue_.empty()) {
                auto h = ready_queue_.front();
                ready_queue_.pop_front();
                if (!h.done()) h.resume();
            }

            // 2. 计算等待时间（取Timer和IO的最小值）
            auto timeout = timer_mgr_.nextTimeout();
            int timeout_ms = std::min((int)timeout.count(), 100);

            // 3. 等待IO事件
            auto io_ready = io_loop_.poll(timeout_ms);
            for (auto h : io_ready) {
                ready_queue_.push_back(h);
            }

            // 4. 检查到期的Timer
            auto timer_ready = timer_mgr_.pollExpired();
            for (auto h : timer_ready) {
                ready_queue_.push_back(h);
            }
        }
    }

    IOLoop& io() { return io_loop_; }
    TimerManager& timers() { return timer_mgr_; }

private:
    std::deque<std::coroutine_handle<>> ready_queue_;
    IOLoop io_loop_;
    TimerManager timer_mgr_;
};
```

### 7. 使用示例

```cpp
// 一个简单的echo server
Task<void> handleClient(EventLoop& loop, int client_fd) {
    char buffer[4096];

    while (true) {
        // 异步等待数据到达
        co_await loop.io().waitReadable(client_fd);

        ssize_t n = ::read(client_fd, buffer, sizeof(buffer));
        if (n <= 0) break;  // 连接关闭

        // 异步等待可写
        co_await loop.io().waitWritable(client_fd);

        ::write(client_fd, buffer, n);  // 回显
    }

    ::close(client_fd);
}

Task<void> acceptLoop(EventLoop& loop, int server_fd) {
    while (true) {
        co_await loop.io().waitReadable(server_fd);

        int client_fd = ::accept4(server_fd, nullptr, nullptr, SOCK_NONBLOCK);
        if (client_fd < 0) continue;

        // 为每个客户端spawcn一个协程（轻量！）
        loop.spawn(handleClient(loop, client_fd));
    }
}

int main() {
    int server_fd = createListenSocket(8080);  // 创建监听socket

    EventLoop loop;
    loop.spawn(acceptLoop(loop, server_fd));
    loop.run();  // 运行事件循环

    return 0;
}

// 每个连接一个协程，同步写法，异步执行
// 10万个并发连接 = 10万个协程（每个只占几百字节栈空间）
// vs 线程模型: 10万个线程 = 10万×8MB = 800GB内存（不可能）
```

### 9. 总结

实现一个协程调度器的核心组件：

| 组件 | 职责 | 关键设计 |
|------|------|---------|
| Task<T> | 协程的返回类型 | initial_suspend + FinalAwaiter实现链式调度 |
| Scheduler | 决定执行哪个协程 | 就绪队列 + 工作窃取 |
| TimerManager | 定时唤醒协程 | 有序集合 + poll到期 |
| IOLoop | 异步IO事件驱动 | epoll + 将协程handle存入event.data |
| EventLoop | 整合以上组件 | poll-dispatch循环 |

协程调度器的本质是一个**事件循环**——不断地从各种事件源（就绪队列、Timer、IO）取出可以恢复的协程并resume它。理解了这个核心循环，无论是asio还是io_uring的协程化封装，都是同一套思路的变体。
