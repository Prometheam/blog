### 引言

CMake是C++世界的事实标准构建系统。但"会写CMake"和"写好CMake"之间差距巨大——我见过太多项目的CMakeLists.txt充斥着全局变量、硬编码路径、不可移植的编译选项，导致在不同平台或不同开发者机器上构建失败。

现代CMake（3.x+）的核心理念是**基于Target的属性传播**——忘掉`include_directories()`和`link_libraries()`这些全局命令，拥抱`target_xxx()`。这篇文章总结我在多个跨平台C++项目中积累的CMake最佳实践。

---

### 1. 现代CMake核心理念

#### 1.1 Target-based思维

```cmake
# ❌ 旧式写法（全局污染）
include_directories(/usr/local/include)
link_directories(/usr/local/lib)
add_definitions(-DUSE_SSL)
set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -Wall")

# ✅ 现代写法（Target级别）
add_library(mylib src/lib.cpp)
target_include_directories(mylib PUBLIC include/)
target_link_libraries(mylib PRIVATE OpenSSL::SSL)
target_compile_definitions(mylib PUBLIC USE_SSL)
target_compile_options(mylib PRIVATE -Wall)
```

#### 1.2 PUBLIC/PRIVATE/INTERFACE

```
PRIVATE   — 只有当前Target使用（编译自身时需要）
PUBLIC    — 当前Target和所有依赖它的Target都使用
INTERFACE — 只有依赖它的Target使用（当前Target不用）

示例：
  mylib 的头文件在 include/，实现用了 boost
  
  target_include_directories(mylib PUBLIC include/)     # 使用者也需要include
  target_link_libraries(mylib PRIVATE Boost::filesystem) # 使用者不需要boost
```

依赖传播：
```
A → B → C

B: target_link_libraries(B PUBLIC A)
   → C自动获得A的PUBLIC属性

B: target_link_libraries(B PRIVATE A)
   → C不会获得A的任何属性
```

#### 根 CMakeLists.txt

```cmake
cmake_minimum_required(VERSION 3.20)

project(MyProject
    VERSION 1.2.3
    DESCRIPTION "A high-performance server"
    LANGUAGES CXX
)

# 仅在顶层项目时启用测试和工具（作为子项目时不启用）
if(CMAKE_PROJECT_NAME STREQUAL PROJECT_NAME)
    set(CMAKE_CXX_STANDARD 20)
    set(CMAKE_CXX_STANDARD_REQUIRED ON)

    enable_testing()
endif()

# 包含CMake模块
list(APPEND CMAKE_MODULE_PATH "${CMAKE_SOURCE_DIR}/cmake")

include(Sanitizers)

# 子目录
add_subdirectory(third_party)
add_subdirectory(src)

if(BUILD_TESTING)
    add_subdirectory(tests)
endif()
```

### 3. 依赖管理

#### 3.1 find_package（系统已安装的库）

```cmake
# 查找OpenSSL
find_package(OpenSSL REQUIRED)
target_link_libraries(mylib PRIVATE OpenSSL::SSL OpenSSL::Crypto)

# 查找Protobuf
find_package(Protobuf REQUIRED)
target_link_libraries(mylib PRIVATE protobuf::libprotobuf)

# 查找线程库（跨平台）
find_package(Threads REQUIRED)
target_link_libraries(mylib PRIVATE Threads::Threads)
```

#### 3.2 FetchContent（自动下载依赖）

```cmake
include(FetchContent)

# 下载并集成 spdlog
FetchContent_Declare(
    spdlog
    GIT_REPOSITORY https://github.com/gabime/spdlog.git
    GIT_TAG        v1.12.0
    GIT_SHALLOW    TRUE  # 浅克隆，加速下载
)

# 下载并集成 googletest
FetchContent_Declare(
    googletest
    GIT_REPOSITORY https://github.com/google/googletest.git
    GIT_TAG        v1.14.0
    GIT_SHALLOW    TRUE
)

# 统一下载（并行）
FetchContent_MakeAvailable(spdlog googletest)

# 使用
target_link_libraries(mylib PRIVATE spdlog::spdlog)
target_link_libraries(mytest PRIVATE GTest::gtest_main)
```

#### 3.3 CPM.cmake（更优雅的包管理）

```cmake
# 引入CPM（只需一个cmake文件）
include(cmake/CPM.cmake)

CPMAddPackage("gh:fmtlib/fmt#10.1.0")
CPMAddPackage("gh:nlohmann/json@3.11.2")
CPMAddPackage(
    NAME benchmark
    GITHUB_REPOSITORY google/benchmark
    VERSION 1.8.3
    OPTIONS "BENCHMARK_ENABLE_TESTING OFF"
)

target_link_libraries(mylib
    PRIVATE
        fmt::fmt
        nlohmann_json::nlohmann_json
)
```

### 4. 编译选项管理

#### 4.1 编译警告（推荐严格模式）

```cmake
# cmake/CompilerWarnings.cmake
function(set_project_warnings target)
    set(MSVC_WARNINGS /W4 /WX /permissive-)
    set(CLANG_WARNINGS
        -Wall -Wextra -Wpedantic
        -Wshadow
        -Wnon-virtual-dtor
        -Wold-style-cast
        -Wcast-align
        -Wunused
        -Woverloaded-virtual
        -Wconversion
        -Wsign-conversion
        -Wnull-dereference
        -Wformat=2
    )
    set(GCC_WARNINGS ${CLANG_WARNINGS}
        -Wmisleading-indentation
        -Wduplicated-cond
        -Wduplicated-branches
        -Wlogical-op
    )

    if(MSVC)
        target_compile_options(${target} PRIVATE ${MSVC_WARNINGS})
    elseif(CMAKE_CXX_COMPILER_ID MATCHES ".*Clang")
        target_compile_options(${target} PRIVATE ${CLANG_WARNINGS})
    elseif(CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
        target_compile_options(${target} PRIVATE ${GCC_WARNINGS})
    endif()
endfunction()
```

#### 4.2 Sanitizers

```cmake
# cmake/Sanitizers.cmake
option(ENABLE_ASAN "Enable AddressSanitizer" OFF)
option(ENABLE_TSAN "Enable ThreadSanitizer" OFF)
option(ENABLE_UBSAN "Enable UndefinedBehaviorSanitizer" OFF)

function(enable_sanitizers target)
    if(ENABLE_ASAN)
        target_compile_options(${target} PRIVATE -fsanitize=address -fno-omit-frame-pointer)
        target_link_options(${target} PRIVATE -fsanitize=address)
    endif()

    if(ENABLE_TSAN)
        target_compile_options(${target} PRIVATE -fsanitize=thread)
        target_link_options(${target} PRIVATE -fsanitize=thread)
    endif()

    if(ENABLE_UBSAN)
        target_compile_options(${target} PRIVATE -fsanitize=undefined)
        target_link_options(${target} PRIVATE -fsanitize=undefined)
    endif()
endfunction()
```

使用：
```bash
cmake -B build -DENABLE_ASAN=ON
cmake --build build
```

### 6. 安装和打包

```cmake
# 安装规则
install(TARGETS mylib
    EXPORT MyProjectTargets
    LIBRARY DESTINATION lib
    ARCHIVE DESTINATION lib
    RUNTIME DESTINATION bin
    INCLUDES DESTINATION include
)

install(DIRECTORY include/myproject
    DESTINATION include
)

# 导出CMake Config（让其他项目find_package能找到我们）
install(EXPORT MyProjectTargets
    FILE MyProjectTargets.cmake
    NAMESPACE MyProject::
    DESTINATION lib/cmake/MyProject
)

# 生成ConfigVersion文件
include(CMakePackageConfigHelpers)
write_basic_package_version_file(
    "${CMAKE_CURRENT_BINARY_DIR}/MyProjectConfigVersion.cmake"
    VERSION ${PROJECT_VERSION}
    COMPATIBILITY SameMajorVersion
)
```

### 7. 常用技巧

#### 7.1 Precompiled Headers（预编译头）

```cmake
# C++20大量include导致编译慢？用PCH加速
target_precompile_headers(mylib PRIVATE
    <vector>
    <string>
    <unordered_map>
    <memory>
    <functional>
    <algorithm>
)
```

#### 7.2 Unity Build（合并编译）

```cmake
# 把多个.cpp合并为一个编译单元，大幅减少编译时间
set_target_properties(mylib PROPERTIES UNITY_BUILD ON)
```

#### 7.3 条件编译

```cmake
option(WITH_GPU "Enable GPU acceleration" OFF)

if(WITH_GPU)
    find_package(CUDA REQUIRED)
    target_sources(mylib PRIVATE src/gpu_impl.cu)
    target_compile_definitions(mylib PUBLIC HAS_GPU=1)
    target_link_libraries(mylib PRIVATE CUDA::cudart)
endif()
```

#### 7.4 生成版本信息

```cmake
# 从Git获取版本
execute_process(
    COMMAND git describe --tags --dirty
    WORKING_DIRECTORY ${CMAKE_SOURCE_DIR}
    OUTPUT_VARIABLE GIT_VERSION
    OUTPUT_STRIP_TRAILING_WHITESPACE
    ERROR_QUIET
)

configure_file(
    ${CMAKE_SOURCE_DIR}/src/version.h.in
    ${CMAKE_BINARY_DIR}/generated/version.h
)
```

```cpp
// src/version.h.in

#define PROJECT_VERSION "@PROJECT_VERSION@"
#define GIT_VERSION "@GIT_VERSION@"
#define BUILD_TIMESTAMP "@BUILD_TIMESTAMP@"
```

### 8. CMake反模式（要避免的写法）

```cmake
# ❌ 全局include路径（污染所有target）
include_directories(${SOME_PATH})

# ❌ 修改全局编译选项
set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -Wall")

# ❌ 用file(GLOB)收集源文件（新增文件不会触发重新配置）
file(GLOB SOURCES "src/*.cpp")

# ❌ 硬编码绝对路径
target_include_directories(mylib PUBLIC /home/user/libs/include)

# ❌ 在库中设置CMAKE_CXX_STANDARD（应该让使用者决定）
set(CMAKE_CXX_STANDARD 17)  # 放在库的CMakeLists中是错的
```

正确做法：
```cmake
# ✅ 用target_xxx代替全局命令
target_include_directories(mylib PUBLIC ${SOME_PATH})

# ✅ 显式列出源文件
target_sources(mylib PRIVATE
    src/engine.cpp
    src/parser.cpp
    src/network.cpp
)

# ✅ 用generator expression做条件编译
target_compile_options(mylib PRIVATE
    $<$<CXX_COMPILER_ID:GNU>:-Wall -Wextra>
    $<$<CXX_COMPILER_ID:MSVC>:/W4>
)

# ✅ 用target属性设置C++标准
target_compile_features(mylib PUBLIC cxx_std_20)
```

### 总结

现代CMake的核心原则：

1. **Target即一切**：所有属性（include路径、编译选项、链接库）都绑定在Target上
2. **传播语义明确**：PUBLIC/PRIVATE/INTERFACE让依赖关系一目了然
3. **不污染全局**：避免全局变量和全局命令
4. **声明式思维**：描述"要什么"而不是"怎么做"

记住：**好的CMakeLists.txt应该像好的代码一样可读**——有清晰的结构、合理的抽象、没有重复。如果你的CMakeLists.txt超过200行，考虑拆分到子目录或抽取为函数/模块。
