class ThreadSafeRefCounted {
public:
    ThreadSafeRefCounted() noexcept      : refCount(0) {}

	//禁用拷贝和移动
	ThreadSafeRefCounted(const ThreadSafeRefCounted&amp;) = delete;
	ThreadSafeRefCounted(ThreadSafeRefCounted&amp;&amp;) = delete;
	ThreadSafeRefCounted&amp; operator=(const ThreadSafeRefCounted&amp;) = delete;
	ThreadSafeRefCounted&amp; operator=(const ThreadSafeRefCounted&amp;&amp;) = delete;

	//增加引用计数
    void addRef() noexcept {
        refCount.fetch_add(1, std::memory_order_relaxed);
    }

	//减少引用计数, 应用计数为1时销毁对象
    void release()   noexcept  {
        if (refCount.fetch_sub(1, std::memory_order_acq_rel) == 1) {
            delete this;
        }
    }

	//获取当前引用计数(主要用于调试)
	int getRefCount() const noexcept{
		return refCount.load(std::memory_order_relaxed);
	}
    
protected:
    virtual ~ThreadSafeRefCounted() = default;
    
private:
	
    std::atomic&lt;unsigned int&gt; refCount;
};


//使用RAII的智能指针包装器
template&lt;typename T&gt;
class RefCountedPtr{
public:
	//构造函数
    explicit RefCountedPtr(T* ptr = nullptr) noexcept : ptr_(ptr) {
		if(ptr_)		{
			ptr_-&gt;addRef();
		}
	}
	
	//拷贝构造函数
	RefCountedPtr(const RefCountedPtr&amp; other) noexcept :ptr_(other.ptr_) {
		if(ptr_)		{
			ptr_-&gt;addRef;
		}
	}
	//移动构造函数
	RefCountedPtr(RefCountedPtr&amp;&amp; other) noexcept : ptr_(other.ptr_){
		other.ptr_ = nullptr;
	}
	
	//析构函数
	~RefCountedPtr()	{
		if(ptr_)		{
			ptr_-&gt;release();
		}
	}
	//拷贝赋值运算符
	RefCountedPtr&amp; operator=(const RefCountedPtr&amp; other) noexcept	{
		if(this != &amp;other)		{
            		// 先增加新对象的引用计数，再减少旧对象的
			if(other.ptr_)			{
				other.ptr_-&gt;addRef();
			}
			if(ptr_)			{
				ptr_-&gt;release();
			}
			ptr_ = other.ptr_;
		}
		return *this;
	}
	
	

	
    	// 移动赋值运算符
	RefCountedPtr&amp; operator=( RefCountedPtr&amp;&amp; other) noexcept{
		if(this != &amp;other)		{
			if(ptr_)			{
				ptr_-&gt;release();
			}
			ptr_ = other.ptr_;
			other.ptr_ = nullptr;
		}
	}


	//解引用运算符
	T&amp; operator*() const noexcept{
		return *ptr_;
	}
	//箭头运算符
	T* operator-&gt;()const noexcept	{
		return ptr_;
	}
	
	//获取原始指针
	T* get() const noexcept{
		return ptr_;
	}
	//重置指针
	void reset(T* ptr = nullptr) noexcept{
		if(ptr_)		{
			ptr_-&gt;release();
		}
		ptr_ = ptr;
		if(ptr_)		{
			ptr_-&gt;addRef;
		}
	}
	//布尔转换
	explicit operator bool() const noexcept{
		return ptr_ != nullptr;
	}
		
	//交换
	void swap(RefCountedPtr&amp; other) noexcept{
		std::swap(ptr_,other.ptr_);
	}
	
private:
	T*	ptr_;
};</code></pre><p style=""></p>
