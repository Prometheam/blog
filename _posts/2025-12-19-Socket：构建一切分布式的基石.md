ss -ant | awk 'NR&gt;1 {++S[$1]} END {for(a in S) print a, S[a]}'</code></pre><p style=""><strong>常见问题：</strong></p><ol><li><p style=""><strong>CLOSE_WAIT过多</strong></p><ul><li><p style="">原因：对方关闭连接后，我方未调用<code>close()</code></p></li><li><p style="">解决方案：检查代码逻辑，确保资源正确释放</p></li></ul></li><li><p style=""><strong>TIME_WAIT过多</strong></p><ul><li><p style="">原因：我方主动关闭连接，等待2MSL（60秒）</p></li><li><p style="">解决方案：调整内核参数或使用<code>SO_REUSEADDR</code></p></li></ul></li></ol><pre><code>int reuse = 1;
setsockopt(sockfd, SOL_SOCKET, SO_REUSEADDR, &amp;reuse, sizeof(reuse));</code></pre><p style=""></p><h3 style="" id="%E4%BA%8C%E3%80%81%E5%85%B3%E9%94%AEapi%EF%BC%9A%E4%BB%8E%E9%98%BB%E5%A1%9E%E5%88%B0%E9%9D%9E%E9%98%BB%E5%A1%9E%E7%9A%84%E6%BC%94%E8%BF%9B">二、关键API：从阻塞到非阻塞的演进</h3><h4 style="" id="2.1-%E9%98%BB%E5%A1%9E%E6%A8%A1%E5%BC%8F%EF%BC%9A%E7%AE%80%E5%8D%95%E7%9A%84%E4%BB%A3%E4%BB%B7">2.1 阻塞模式：简单的代价</h4><pre><code>// 阻塞式读数据
char buffer[1024];
int n = read(sockfd, buffer, sizeof(buffer)); // 阻塞直到有数据或连接关闭</code></pre><p style="text-indent: 2em"><span style="font-size: 16px; color: rgb(15, 17, 21)">阻塞模式简单直观，但一个线程只能处理一个连接，无法应对高并发场景。在C10K问题提出后，这种模式逐渐被淘汰。</span></p><h4 style="" id="2.2-%E9%9D%9E%E9%98%BB%E5%A1%9E%E6%A8%A1%E5%BC%8F%EF%BC%9A%E9%AB%98%E5%B9%B6%E5%8F%91%E7%9A%84%E4%BB%A3%E4%BB%B7">2.2 非阻塞模式：高并发的代价</h4><pre><code>// 设置为非阻塞模式
int flags = fcntl(sockfd, F_GETFL, 0);
fcntl(sockfd, F_SETFL, flags | O_NONBLOCK);

// 非阻塞读取
int n = read(sockfd, buffer, sizeof(buffer));
if (n &lt; 0) {
    if (errno == EAGAIN || errno == EWOULDBLOCK) {
        // 没有数据，稍后重试
    } else {
        // 真正的错误
    }
}</code></pre><p style="text-indent: 2em"><span style="font-size: 16px; color: rgb(15, 17, 21)">核心是 </span><code>fcntl(sockfd, F_SETFL, O_NONBLOCK)</code><span style="font-size: 16px; color: rgb(15, 17, 21)">。</span><code>accept</code><span style="font-size: 16px; color: rgb(15, 17, 21)">、</span><code>connect</code><span style="font-size: 16px; color: rgb(15, 17, 21)">、</span><code>recv</code><span style="font-size: 16px; color: rgb(15, 17, 21)">、</span><code>send</code><span style="font-size: 16px; color: rgb(15, 17, 21)"> 会立即返回，通过 </span><code>errno</code><span style="font-size: 16px; color: rgb(15, 17, 21)">（</span><code>EAGAIN</code><span style="font-size: 16px; color: rgb(15, 17, 21)">/</span><code>EWOULDBLOCK</code><span style="font-size: 16px; color: rgb(15, 17, 21)">）判断状态。</span><strong>这是实现高并发 I/O 多路复用的前提</strong><span style="font-size: 16px; color: rgb(15, 17, 21)">。</span></p><p style=""></p><h3 style="" id="%E4%B8%89%E3%80%81i%2Fo%E5%A4%9A%E8%B7%AF%E5%A4%8D%E7%94%A8%EF%BC%9A%E9%AB%98%E5%B9%B6%E5%8F%91%E6%9C%8D%E5%8A%A1%E5%99%A8%E7%9A%84%E6%A0%B8%E5%BF%83%E6%8A%80%E6%9C%AF">三、I/O多路复用：高并发服务器的核心技术</h3><h4 style="" id="3.1-select%EF%BC%9A%E6%9C%80%E5%8F%A4%E8%80%81%E7%9A%84%E5%A4%9A%E8%B7%AF%E5%A4%8D%E7%94%A8">3.1 select：最古老的多路复用</h4><p style=""><a href="http://124.220.188.239/console/posts/editor?name=d1a0ce66-8c9f-4edf-bfac-e4ab448ba642" target="_blank" rel="">完整代码示例</a></p><pre><code>fd_set readfds;
FD_ZERO(&amp;readfds);
FD_SET(sockfd, &amp;readfds);

struct timeval timeout = {5, 0}; // 5秒超时
int ready = select(sockfd + 1, &amp;readfds, NULL, NULL, &amp;timeout);</code></pre><p style="text-indent: 2em"><strong>select的局限性：</strong>监听fd数量限制（通常1024）、每次调用需要复制整个fd_set到内核、返回后需要线性扫描所有fd。</p><p style="">△那么问题来了：<code>select</code> 返回后，如何高效找出就绪的 fd？<br>		这是<code>select</code><span style="font-size: 16px; color: rgb(15, 17, 21)"> 模型一个主要的性能瓶颈。</span><code>select</code><span style="font-size: 16px; color: rgb(15, 17, 21)"> 返回后，我们需要遍历</span><strong>整个</strong><span style="font-size: 16px; color: rgb(15, 17, 21)">被监控的 fd 集合（通常是三个集合：读、写、异常）来找出哪些 fd 就绪了。这个过程是 </span><strong>O(n)</strong><span style="font-size: 16px; color: rgb(15, 17, 21)"> 的，效率低下。</span></p><p style="text-indent: 2em"><span style="font-size: 16px; color: rgb(15, 17, 21)">核心问题在于：</span><code>select</code><strong> </strong>只告诉你有事件发生，但没有直接告诉你哪些 fd 就绪，你需要自己<strong>用 </strong><code>FD_ISSET</code><strong> 宏去轮询检查。</strong></p><ul><li><p style="">策略1（低效）：<span style="font-size: 16px; color: rgb(15, 17, 21)">这是最直接的写法，也是教科书上常见的，但在高并发下是性能杀手</span></p></li></ul><pre><code>//缺点：每次都需要从 0 遍历到 maxfd（通常是数百甚至上千），即使只有1个fd就绪。
fd_set readfds_copy = master_readfds; // 每次调用select前需要复制
int ret = select(maxfd+1, &amp;readfds_copy, NULL, NULL, NULL);

for (int fd = 0; fd &lt;= maxfd; fd++) {
    if (FD_ISSET(fd, &amp;readfds_copy)) {
        // 处理 fd 上的读事件
    }
}</code></pre><ul><li><p style="">策略2（相对高效）：<strong>维护独立的“就绪fd列表”或“活跃fd列表”</strong></p></li></ul><p style="text-indent: 2em"><span style="font-size: 16px; color: rgb(15, 17, 21)">不要每次都遍历所有可能的 fd，而是只遍历你</span><strong>真正关心</strong><span style="font-size: 16px; color: rgb(15, 17, 21)">的 fd。</span></p><pre><code>/* 
实现：用一个数组、链表或更高效的结构（如红黑树）来只保存你通过 FD_SET 加入到 select 监控集合中的 fd。

过程：
你有一个 active_fd_list，里面是所有被监控的 fd。
select 返回后，你只遍历这个 active_fd_list，对列表中的每个 fd 使用 FD_ISSET 检查。

优点：
遍历次数 = 当前活跃连接数，而不是最大fd值。对于 maxfd=1024 但只有10个连接的情况，遍历次数从1024次降到10次。
这是最有效、最常用的优化手段。
*/

// 假设我们用一个数组来管理所有被监控的客户端fd
int client_fds[MAX_CLIENTS];
int client_count = 0;

// ... 有新的客户端连接时，将fd加入到client_fds和readfds中 ...

// select 返回后
for (int i = 0; i &lt; client_count; i++) {
    int fd = client_fds[i];
    if (FD_ISSET(fd, &amp;readfds_copy)) {
        // 处理这个客户端的读事件
    }
    // 注意：如果连接关闭，需要从client_fds中移除（后续可能需要压缩数组）
}</code></pre><ul><li><p style="">策略3（相对高效）：利用select的返回值</p></li></ul><p style="text-indent: 2em"><code>select</code><span style="font-size: 16px; color: rgb(15, 17, 21)"> 的返回值 </span><code>nready</code><span style="font-size: 16px; color: rgb(15, 17, 21)"> 是</span><strong>总共就绪的 fd 数量</strong><span style="font-size: 16px; color: rgb(15, 17, 21)">。可以在循环中利用这个值提前退出。</span></p><pre><code>/* 
实现：在遍历过程中，每处理一个就绪的 fd，就将 nready 减1。当 nready 减到 0 时，说明所有就绪事件都已处理，立即跳出循环，无需检查剩余的 fd。

优点：在就绪事件很少时，能显著减少不必要的 FD_ISSET 调用。
*/

int nready = select(maxfd+1, &amp;readfds_copy, NULL, NULL, NULL);
for (int fd = 0; fd &lt;= maxfd &amp;&amp; nready &gt; 0; fd++) {
    if (FD_ISSET(fd, &amp;readfds_copy)) {
        nready--;
        // 处理 fd
    }
}</code></pre><p style=""></p><ul><li><p style="">策略4（适用于超大规模）：分组与分层</p></li></ul><p style="text-indent: 2em"><span style="font-size: 16px; color: rgb(15, 17, 21)">当连接数巨大时（例如数万），即使只遍历活跃连接列表，开销也可能很大。可以考虑将连接分组，使用多个 </span><code>select</code><span style="font-size: 16px; color: rgb(15, 17, 21)"> 线程或进程，每个负责一个子集（类似 </span><code>SO_REUSEPORT</code><span style="font-size: 16px; color: rgb(15, 17, 21)"> 的思想）。但这已经超出了单纯优化 </span><code>select</code><span style="font-size: 16px; color: rgb(15, 17, 21)"> 本身的范围，属于架构层面的优化。</span></p><p style="text-indent: 2em"></p><ul><li><p style="">策略5：<strong>内核态与用户态的考量</strong></p></li></ul><p style="text-indent: 2em"><code>FD_ISSET</code><span style="font-size: 16px; color: rgb(15, 17, 21)"> 是一个宏，它检查一个 </span><code>fd_set</code><span style="font-size: 16px; color: rgb(15, 17, 21)">（本质是位图）中的某一位。这个检查本身非常快（位操作）。主要的开销来自于</span><strong>遍历这个行为</strong><span style="font-size: 16px; color: rgb(15, 17, 21)">，以及</span><strong>遍历过程中可能发生的缓存不命中</strong><span style="font-size: 16px; color: rgb(15, 17, 21)">。策略一（只遍历活跃列表）能很好地改善缓存局部性。</span></p><h4 style="" id="3.2-poll%EF%BC%9A%E6%94%B9%E8%BF%9B%E5%8D%95%E6%98%AF%E4%BB%8D%E6%9C%89%E9%99%90%E5%88%B6">3.2 poll：改进单是仍有限制</h4><p style=""><a href="http://124.220.188.239/console/posts/editor?name=e8f5d638-97e3-4f32-a958-1d5b9f583dbe" target="_blank" rel="">完整代码示例</a></p><pre><code>struct pollfd fds[1];
fds[0].fd = sockfd;
fds[0].events = POLLIN;

int ready = poll(fds, 1, 5000); // 5秒超时</code></pre><p style="text-indent: 2em"><span style="font-size: 16px; color: rgb(15, 17, 21)">poll解决了select的fd数量限制（使用链表，无最大 fd 限制），但本质上仍是O(n)的轮询机制。</span></p><p style=""></p><h4 style="" id="3.3-epoll-linux%E7%9A%84%E7%BB%88%E6%9E%81%E6%AD%A6%E5%99%A8">3.3 epoll Linux的终极武器</h4><p style=""><a href="http://124.220.188.239/console/posts/editor?name=91f55f13-630b-4fbb-9480-04712591f478" target="_blank" rel="">完整代码示例</a></p><pre><code>// 创建epoll实例
int epoll_fd = epoll_create1(0);

// 添加监听socket
struct epoll_event event;
event.events = EPOLLIN | EPOLLET; // 边缘触发模式
event.data.fd = sockfd;
epoll_ctl(epoll_fd, EPOLL_CTL_ADD, sockfd, &amp;event);

// 等待事件
struct epoll_event events[10];
int n = epoll_wait(epoll_fd, events, 10, -1);</code></pre><p style=""><strong>epoll的核心优势：</strong></p><ol><li><p style=""><strong>内核数据结构托管</strong>：<code>epoll_create</code> 创建上下文，<code>epoll_ctl</code> 管理 fd，避免每次系统调用传递全部 fd。</p></li><li><p style=""><strong>事件驱动</strong>：<code>epoll_wait</code><span style="font-size: 16px; color: rgb(15, 17, 21)"> 直接返回就绪的 fd 列表，O(1) 复杂度。</span></p></li><li><p style=""><strong>两种触发模式</strong>：</p><ul><li><p style=""><strong>水平触发（LT）</strong>：默认模式，<span style="font-size: 16px; color: rgb(15, 17, 21)">只要 fd 可读/可写，就会持续通知。编程更简单，但若未一次性处理完，会频繁触发。</span></p></li><li><p style=""><strong>边缘触发（ET）</strong>：<span style="font-size: 16px; color: rgb(15, 17, 21)">仅在 fd 状态变化时通知一次。</span><strong>必须使用非阻塞 socket，并且必须循环 read/write 直到 </strong><code>EAGAIN</code><span style="font-size: 16px; color: rgb(15, 17, 21)">，否则会遗漏事件。性能更高，是高性能服务器的标准选择。</span></p></li></ul></li></ol><pre><code>// ET模式的最佳实践：
// ET模式下必须读到EAGAIN
while (true) {
    ssize_t n = read(fd, buf, sizeof(buf));
    if (n &lt; 0) {
        if (errno == EAGAIN || errno == EWOULDBLOCK) {
            break; // 数据读完
        }
        // 处理错误
        break;
    } else if (n == 0) {
        // 连接关闭
        close(fd);
        break;
    }
    // 处理数据
    process_data(buf, n);
}</code></pre><p style="">△ET 模式下，为什么必须读到 <code>EAGAIN</code>？</p><p style="text-indent: 2em">1.要解释这个问题，需要理解ET模式的工作原理：</p><ul><li><p style=""><strong>边缘触发</strong>：只有当文件描述符的<strong>状态发生变化</strong>时（例如从“不可读”变为“可读”，或从“不可写”变为“可写”），才会通知一次事件。</p></li><li><p style=""><strong>一次触发</strong>：如果缓冲区中有数据可读，但你没有一次性读完，<strong>剩余的数据不会再次触发读事件</strong>（除非有新的数据到达，导致状态再次变化）。</p></li></ul><p style="text-indent: 2em">2.现在再来看为什么必须督导<code>EAGAIN</code>？<span style="font-size: 16px; color: rgb(15, 17, 21)">在 ET 模式下，</span><strong>一旦触发读事件，你必须尽可能多地读取数据</strong><span style="font-size: 16px; color: rgb(15, 17, 21)">，直到：</span></p><ul><li><p style="">缓冲区被清空（即 <code>read</code> 返回 <code>EAGAIN</code>，表示没有更多数据可读）。</p></li><li><p style="">如果只读一次就停止，<strong>剩余的数据会留在缓冲区中，且不会再次触发事件</strong>，导致数据滞留，甚至死锁。</p></li></ul><pre><code>// 非阻塞 socket + ET 模式
//read 返回 0 表示对端关闭连接（EOF）。
//返回 -1 且 errno == EAGAIN（或 EWOULDBLOCK）表示当前没有数据可读，但连接正常（非阻塞模式下）。
while (1) {
    ssize_t n = read(fd, buf, sizeof(buf));
    if (n &gt; 0) {
        // 处理读取的数据
        process_data(buf, n);
    } else if (n == 0) {
        // 对端关闭连接
        close(fd);
        break;
    } else if (n == -1) {
        if (errno == EAGAIN || errno == EWOULDBLOCK) {
            // 数据已读完，退出循环
            break;
        } else {
            // 其他错误处理
            perror("read error");
            close(fd);
            break;
        }
    }
}</code></pre><p style="text-indent: 2em">3.顺带说一下LT的工作模式：</p><ul><li><p style=""><strong>LT 模式</strong>：只要缓冲区有数据，就会持续触发读事件，因此你可以一次只读部分数据，下次事件循环会再次通知。</p></li></ul><p style="">△如果使用多线程处理一个 <code>epoll_fd</code>，惊群问题如何产生？如何解决？（Linux 3.9+ 的 <code>EPOLLEXCLUSIVE</code>）</p><p style="text-indent: 2em">1.什么是惊群问题？</p><p style="text-indent: 2em"><strong>惊群问题</strong><span style="font-size: 16px; color: rgb(15, 17, 21)">指的是当多个进程或线程同时等待同一个资源时，一旦该资源就绪，所有等待者都会被唤醒，但最终只有一个能成功获取资源，其他等待者再次陷入等待状态。这种不必要的唤醒造成了大量的上下文切换和 CPU 资源浪费。</span></p><p style="text-indent: 2em">2.在多线程epoll中的表现：</p><p style="text-indent: 2em">主线程创建监听 socket 并注册到 epoll 实例；多个工作线程都调用 <code>epoll_wait()</code> 监听同一个 epoll 文件描述符；当新连接到达时，所有线程都被唤醒；只有一个线程能成功 <code>accept()</code> 这个连接，其他线程唤醒后发现无事可做，再次阻塞</p><p style="text-indent: 2em">3.解决方案1：Linux 3.9+ 的救星：EPOLLEXCLUSIVE </p><p style="text-indent: 2em"><span style="font-size: 16px; color: rgb(15, 17, 21)">Linux 3.9 内核引入了 </span><code>EPOLLEXCLUSIVE</code><span style="font-size: 16px; color: rgb(15, 17, 21)"> 标志，专门用于解决 epoll 的惊群问题。这个标志确保事件发生时，只有一个等待线程被唤醒。</span></p><p style="text-indent: 2em"><strong>工作原理：</strong>当使用 <code>EPOLLEXCLUSIVE</code> 标志将一个文件描述符添加到 epoll 实例时：</p><p style="margin-left: 24px!important; text-indent: 2em">独占唤醒：事件发生时，内核保证只唤醒一个正在等待的线程</p><p style="margin-left: 24px!important; text-indent: 2em">负载均衡：内核会以轮询方式在不同事件上选择不同的线程唤醒，避免饥饿</p><p style="margin-left: 24px!important; text-indent: 2em">向后兼容：不影响未使用此标志的现有代码</p><pre><code>// 创建 epoll 实例
int epoll_fd = epoll_create1(0);

// 准备监听 socket
int listen_fd = socket(AF_INET, SOCK_STREAM, 0);
// ... 绑定、监听等操作

// 使用 EPOLLEXCLUSIVE 标志添加监听 socket
struct epoll_event event;
event.events = EPOLLIN | EPOLLEXCLUSIVE;  // 关键在这里
event.data.fd = listen_fd;

if (epoll_ctl(epoll_fd, EPOLL_CTL_ADD, listen_fd, &amp;event) == -1) {
    perror("epoll_ctl");
    exit(EXIT_FAILURE);
}

// 创建多个工作线程，它们都等待同一个 epoll_fd
for (int i = 0; i &lt; thread_count; i++) {
    pthread_create(&amp;threads[i], NULL, worker_thread, (void*)epoll_fd);
}</code></pre><p style="text-indent: 2em">4.解决方案2：SO_REUSEPORT（端口重用）</p><p style="text-indent: 2em"><code>SO_REUSEPORT</code><span style="font-size: 16px; color: rgb(15, 17, 21)"> 是 Linux 3.9+ 引入的 socket 选项，它允许多个进程或线程绑定到</span><strong>同一个 IP 和端口</strong><span style="font-size: 16px; color: rgb(15, 17, 21)">。内核会使用一个哈希算法将传入连接分发到不同的 socket，从而实现负载均衡。</span></p><p style="text-indent: 2em"></p><h4 style="" id="%E5%9B%9B%E3%80%81%E9%AB%98%E6%80%A7%E8%83%BD%E6%9C%8D%E5%8A%A1%E5%99%A8%E6%9E%B6%E6%9E%84%E6%A8%A1%E5%BC%8F">四、高性能服务器架构模式</h4><h4 style="" id="4.1-reactor%E6%A8%A1%E5%BC%8F%EF%BC%9A%E4%BA%8B%E4%BB%B6%E9%A9%B1%E5%8A%A8%E7%9A%84%E6%A0%87%E5%87%86%E5%AE%9E%E7%8E%B0">4.1 Reactor模式：事件驱动的标准实现</h4><p style="text-indent: 2em"><span style="font-size: 16px; color: rgb(15, 17, 21)">Reactor模式的核心思想是</span><strong>分离I/O事件与业务逻辑。</strong></p><p style="text-indent: 2em"><strong>单Reactor单线程（Redis模式）：</strong></p><pre><code>┌─────────────────┐
