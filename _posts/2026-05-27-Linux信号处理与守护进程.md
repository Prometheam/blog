### 引言

信号（Signal）是Linux进程间通信的最古老方式之一，也是最容易写出Bug的领域。一个看似正常运行的服务，因为信号处理不当，可能在收到SIGPIPE时莫名退出，或者在SIGCHLD处理中死锁。

在我维护的生产服务中，因为信号处理Bug导致的线上事故不止一次。今天系统性地复盘信号处理的正确姿势，以及如何实现一个健壮的守护进程。

---

### 1. 信号基础

#### 1.1 常见信号

```
┌──────────┬──────┬──────────────────────────────────────────┐
│  信号    │ 编号 │           默认行为 / 用途                │
├──────────┼──────┼──────────────────────────────────────────┤
│ SIGHUP   │  1   │ 终止。惯例：重载配置                    │
│ SIGINT   │  2   │ 终止。Ctrl+C                            │
│ SIGQUIT  │  3   │ 终止+core。Ctrl+\                       │
│ SIGKILL  │  9   │ 终止。不可捕获/忽略                     │
│ SIGPIPE  │ 13   │ 终止。向已关闭的socket写数据            │
│ SIGALRM  │ 14   │ 终止。alarm()定时器                     │
│ SIGTERM  │ 15   │ 终止。优雅关闭的标准信号                │
│ SIGCHLD  │ 17   │ 忽略。子进程状态变化                    │
│ SIGSTOP  │ 19   │ 停止。不可捕获                          │
│ SIGUSR1  │ 10   │ 终止。用户自定义                        │
│ SIGUSR2  │ 12   │ 终止。用户自定义                        │
└──────────┴──────┴──────────────────────────────────────────┘
```

#### 1.2 信号的异步本质

信号是**异步的中断**——它可以在程序执行的**任意时刻**打断当前代码，跳转到信号处理函数。这意味着：

```
正常执行流：      main() → foo() → bar() → ...
                                    ↑
信号到达：                    中断！跳转到signal_handler()
                                    ↓
处理完后：                    返回bar()继续执行
```

这带来了严重的**可重入性**问题。

### 2. 信号处理的正确姿势

#### 2.1 信号安全函数（Async-Signal-Safe）

在信号处理函数中，**只能调用异步信号安全的函数**。以下函数是**不安全的**：

```cpp
// ❌ 信号处理函数中禁止调用的函数
printf()        // 内部有锁（stdout mutex）
malloc()/free() // 内部有锁（堆管理器）
new/delete      // 底层调用malloc
std::cout       // 内部有锁+缓冲
syslog()        // 可能调用malloc
pthread_mutex_lock() // 如果主流程正持有同一把锁 → 死锁
```

安全的函数列表（POSIX标准）：
```
write(), read(), close(), _exit()
open(), dup2(), pipe(), fork(), execve()
signal(), sigaction(), sigprocmask()
kill(), raise()
getpid(), getuid()
```

#### 2.2 最安全的信号处理方式：只设Flag

```cpp
// ✅ 推荐：信号处理函数只设一个标志位
volatile sig_atomic_t g_shutdown_requested = 0;
volatile sig_atomic_t g_reload_config = 0;

void signalHandler(int sig) {
    switch (sig) {
        case SIGTERM:
        case SIGINT:
            g_shutdown_requested = 1;
            break;
        case SIGHUP:
            g_reload_config = 1;
            break;
    }
    // 不做任何其他操作！
}

// 主循环中检查标志
void mainLoop() {
    while (!g_shutdown_requested) {
        if (g_reload_config) {
            g_reload_config = 0;
            loadConfig();  // 安全：在主线程中执行
        }
        // 正常业务逻辑...
    }
    gracefulShutdown();
}
```

#### 2.3 Self-Pipe技巧

问题：主循环阻塞在epoll_wait时，收到信号后需要立即唤醒。

```cpp
// Self-Pipe：信号处理函数往pipe写一字节，epoll监听pipe读端
int g_signal_pipe[2];  // [0]=读端, [1]=写端

void signalHandler(int sig) {
    int saved_errno = errno;  // 保存errno
    char c = static_cast<char>(sig);
    write(g_signal_pipe[1], &c, 1);  // write是信号安全的
    errno = saved_errno;  // 恢复errno
}

void setupSignalPipe() {
    pipe2(g_signal_pipe, O_NONBLOCK | O_CLOEXEC);
    
    // 将pipe读端加入epoll
    struct epoll_event ev{EPOLLIN, {.fd = g_signal_pipe[0]}};
    epoll_ctl(epfd, EPOLL_CTL_ADD, g_signal_pipe[0], &ev);
}

// 事件循环中：
void eventLoop() {
    struct epoll_event events[64];
    while (true) {
        int n = epoll_wait(epfd, events, 64, -1);
        for (int i = 0; i < n; ++i) {
            if (events[i].data.fd == g_signal_pipe[0]) {
                // 信号到达，读取信号编号
                char sig;
                while (read(g_signal_pipe[0], &sig, 1) > 0) {
                    handleSignal(static_cast<int>(sig));
                }
            } else {
                // 正常IO事件处理...
            }
        }
    }
}
```

#### 2.4 signalfd — 更现代的方案（Linux 2.6.22+）

```cpp
#include <sys/signalfd.h>

void setupSignalfd() {
    sigset_t mask;
    sigemptyset(&mask);
    sigaddset(&mask, SIGTERM);
    sigaddset(&mask, SIGINT);
    sigaddset(&mask, SIGHUP);
    
    // 阻塞这些信号（不触发默认处理）
    sigprocmask(SIG_BLOCK, &mask, nullptr);
    
    // 创建signalfd
    int sfd = signalfd(-1, &mask, SFD_NONBLOCK | SFD_CLOEXEC);
    
    // 加入epoll
    struct epoll_event ev{EPOLLIN, {.fd = sfd}};
    epoll_ctl(epfd, EPOLL_CTL_ADD, sfd, &ev);
}

// 事件循环中读取信号
void handleSignalfd(int sfd) {
    struct signalfd_siginfo info;
    while (read(sfd, &info, sizeof(info)) == sizeof(info)) {
        switch (info.ssi_signo) {
            case SIGTERM: gracefulShutdown(); break;
            case SIGHUP:  reloadConfig(); break;
        }
    }
}
```

**优势**：信号变成了普通的文件描述符事件，可以和IO事件统一处理，无需担心可重入问题。

### 3. 必须处理的信号

#### 3.1 SIGPIPE — 最容易忽略的杀手

```cpp
// 问题：向已关闭的TCP连接写数据，内核发送SIGPIPE
// 默认行为：进程退出！

// 解决方案1：全局忽略（推荐）
signal(SIGPIPE, SIG_IGN);

// 解决方案2：使用MSG_NOSIGNAL标志
send(fd, data, len, MSG_NOSIGNAL);  // 不产生SIGPIPE，send返回-1/EPIPE

// 解决方案3：对写端shutdown后检查
// write返回-1且errno==EPIPE时处理断开
```

**工程经验**：每个网络服务启动时第一行就应该`signal(SIGPIPE, SIG_IGN)`。

#### 3.2 SIGCHLD — 避免僵尸进程

```cpp
// 子进程退出后如果父进程不wait()，子进程变为僵尸进程（zombie）
// 解决方案：
void sigchldHandler(int) {
    int saved_errno = errno;
    // waitpid(-1, ...) 收割所有已终止子进程
    while (waitpid(-1, nullptr, WNOHANG) > 0);
    errno = saved_errno;
}

// 或者直接忽略（Linux特有，自动回收）
signal(SIGCHLD, SIG_IGN);
```

### 4. 守护进程（Daemon）实现

守护进程是脱离终端、在后台长期运行的服务进程。标准的daemonize流程：

```cpp
#include <unistd.h>
#include <sys/stat.h>
#include <fcntl.h>
#include <cstdlib>

bool daemonize() {
    // 1. 第一次fork：脱离终端
    pid_t pid = fork();
    if (pid < 0) return false;
    if (pid > 0) _exit(0);  // 父进程退出，子进程被init收养

    // 2. 创建新会话（脱离原会话和控制终端）
    if (setsid() < 0) return false;

    // 3. 第二次fork：确保不是会话首进程（永远不会获得控制终端）
    pid = fork();
    if (pid < 0) return false;
    if (pid > 0) _exit(0);

    // 4. 设置文件掩码（确保daemon创建的文件有合适权限）
    umask(0022);

    // 5. 切换工作目录（避免占用可卸载的文件系统）
    chdir("/");

    // 6. 关闭标准IO，重定向到/dev/null
    close(STDIN_FILENO);
    close(STDOUT_FILENO);
    close(STDERR_FILENO);
    open("/dev/null", O_RDONLY);  // stdin  → /dev/null (fd=0)
    open("/dev/null", O_WRONLY);  // stdout → /dev/null (fd=1)
    open("/dev/null", O_WRONLY);  // stderr → /dev/null (fd=2)

    // 7. 写入PID文件（防止重复启动）
    writePidFile("/var/run/myservice.pid");

    return true;
}

void writePidFile(const char* path) {
    int fd = open(path, O_WRONLY | O_CREAT | O_TRUNC, 0644);
    if (fd >= 0) {
        char buf[32];
        int len = snprintf(buf, sizeof(buf), "%d\n", getpid());
        write(fd, buf, len);
        close(fd);
    }
}
```

#### 4.1 现代Linux替代方案：systemd

现代Linux发行版推荐用systemd管理服务，无需手动daemonize：

```ini
# /etc/systemd/system/myservice.service
[Unit]
Description=My C++ Service
After=network.target

[Service]
Type=simple            # 不需要daemonize！systemd直接管理前台进程
ExecStart=/usr/bin/myservice --config /etc/myservice.conf
ExecReload=/bin/kill -HUP $MAINPID
Restart=always
RestartSec=5
User=myservice
LimitNOFILE=1048576

[Install]
WantedBy=multi-user.target
```

```bash
systemctl enable myservice
systemctl start myservice
systemctl status myservice
journalctl -u myservice -f  # 查看日志
```

### 5. 完整示例：健壮的服务进程

```cpp
#include <signal.h>
#include <sys/signalfd.h>
#include <sys/epoll.h>
#include <unistd.h>
#include <cstdio>
#include <cstdlib>

class ServiceProcess {
public:
    bool init() {
        // 忽略SIGPIPE
        signal(SIGPIPE, SIG_IGN);

        // 用signalfd处理终止和重载信号
        sigset_t mask;
        sigemptyset(&mask);
        sigaddset(&mask, SIGTERM);
        sigaddset(&mask, SIGINT);
        sigaddset(&mask, SIGHUP);
        sigaddset(&mask, SIGUSR1);
        sigprocmask(SIG_BLOCK, &mask, nullptr);

        signal_fd_ = signalfd(-1, &mask, SFD_NONBLOCK | SFD_CLOEXEC);
        if (signal_fd_ < 0) return false;

        epoll_fd_ = epoll_create1(EPOLL_CLOEXEC);
        if (epoll_fd_ < 0) return false;

        // 注册signalfd到epoll
        struct epoll_event ev{EPOLLIN, {.fd = signal_fd_}};
        epoll_ctl(epoll_fd_, EPOLL_CTL_ADD, signal_fd_, &ev);

        return true;
    }

    void run() {
        printf("[Service] Started, PID=%d\n", getpid());
        struct epoll_event events[64];

        while (running_) {
            int n = epoll_wait(epoll_fd_, events, 64, 1000);

            if (n < 0) {
                if (errno == EINTR) continue;
                break;
            }

            for (int i = 0; i < n; ++i) {
                if (events[i].data.fd == signal_fd_) {
                    handleSignals();
                } else {
                    handleIO(events[i]);
                }
            }

            // 定时任务（每秒执行）
            doPeriodicWork();
        }

        cleanup();
        printf("[Service] Stopped gracefully\n");
    }

private:
    void handleSignals() {
        struct signalfd_siginfo info;
        while (read(signal_fd_, &info, sizeof(info)) == sizeof(info)) {
            switch (info.ssi_signo) {
                case SIGTERM:
                case SIGINT:
                    printf("[Service] Received signal %d, shutting down...\n",
                           info.ssi_signo);
                    running_ = false;
                    break;
                case SIGHUP:
                    printf("[Service] Reloading config...\n");
                    reloadConfig();
                    break;
                case SIGUSR1:
                    printf("[Service] Dumping stats...\n");
                    dumpStats();
                    break;
            }
        }
    }

    void handleIO(const epoll_event& ev) { /* ... */ }
    void doPeriodicWork() { /* ... */ }
    void reloadConfig() { /* ... */ }
    void dumpStats() { /* ... */ }
    void cleanup() { /* ... */ }

    int signal_fd_ = -1;
    int epoll_fd_ = -1;
    bool running_ = true;
};

int main(int argc, char* argv[]) {
    // 可选：以守护进程方式运行
    if (argc > 1 && strcmp(argv[1], "-d") == 0) {
        daemonize();
    }

    ServiceProcess service;
    if (!service.init()) {
        fprintf(stderr, "Failed to initialize\n");
        return 1;
    }
    service.run();
    return 0;
}
```

### 6. 信号处理的常见Bug

| Bug | 原因 | 解决 |
|-----|------|------|
| 信号处理中调printf死锁 | printf持有stdout锁，信号中断了printf | 只用write() |
| EINTR未处理导致逻辑错误 | 慢系统调用被信号中断返回-1 | 检查errno==EINTR重试 |
| 多线程中信号送到错误线程 | 信号送给进程中的任意线程 | pthread_sigmask阻塞+专用信号线程 |
| 僵尸进程堆积 | 未处理SIGCHLD | waitpid回收 |
| 进程莫名退出 | SIGPIPE未忽略 | signal(SIGPIPE, SIG_IGN) |

### 总结

信号处理的黄金法则：
1. **处理函数越简单越好**（最好只设标志位或写pipe）
2. **全局忽略SIGPIPE**
3. **优先用signalfd替代signal handler**
4. **多线程环境用专用线程处理信号**
5. **生产服务用systemd管理，不手写daemonize**
