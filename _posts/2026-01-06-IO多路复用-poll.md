<h4 style="" id="c%2B%2B%E8%AF%AD%E8%A8%80%E5%AE%8C%E6%95%B4%E5%AE%9E%E7%8E%B0">C++语言完整实现</h4><pre><code>#include &lt;iostream&gt;
#include &lt;vector&gt;
#include &lt;string&gt;
#include &lt;cstring&gt;
#include &lt;unistd.h&gt;
#include &lt;sys/socket.h&gt;
#include &lt;netinet/in.h&gt;
#include &lt;arpa/inet.h&gt;
#include &lt;poll.h&gt;
#include &lt;fcntl.h&gt;
#include &lt;errno.h&gt;

class PollServer {
private:
    int server_fd;
    int port;
    std::vector&lt;pollfd&gt; fds;
    
public:
    PollServer(int port) : port(port), server_fd(-1) {}
    
    ~PollServer() {
        cleanup();
    }
    
    bool initialize() {
        // 创建socket
        server_fd = socket(AF_INET, SOCK_STREAM, 0);
        if (server_fd &lt; 0) {
            std::cerr &lt;&lt; "创建socket失败: " &lt;&lt; strerror(errno) &lt;&lt; std::endl;
            return false;
        }
        
        // 设置socket选项，允许地址重用
        int opt = 1;
        if (setsockopt(server_fd, SOL_SOCKET, SO_REUSEADDR, &amp;opt, sizeof(opt)) &lt; 0) {
            std::cerr &lt;&lt; "设置socket选项失败: " &lt;&lt; strerror(errno) &lt;&lt; std::endl;
            return false;
        }
        
        // 绑定地址
        struct sockaddr_in addr;
        memset(&amp;addr, 0, sizeof(addr));
        addr.sin_family = AF_INET;
        addr.sin_addr.s_addr = INADDR_ANY;
        addr.sin_port = htons(port);
        
        if (bind(server_fd, (struct sockaddr*)&amp;addr, sizeof(addr)) &lt; 0) {
            std::cerr &lt;&lt; "绑定地址失败: " &lt;&lt; strerror(errno) &lt;&lt; std::endl;
            return false;
        }
        
        // 监听连接
        if (listen(server_fd, 10) &lt; 0) {
            std::cerr &lt;&lt; "监听失败: " &lt;&lt; strerror(errno) &lt;&lt; std::endl;
            return false;
        }
        
        // 设置非阻塞模式（可选）
        fcntl(server_fd, F_SETFL, fcntl(server_fd, F_GETFL, 0) | O_NONBLOCK);
        
        // 将server_fd添加到pollfd数组
        pollfd server_pollfd;
        server_pollfd.fd = server_fd;
        server_pollfd.events = POLLIN;  // 监听可读事件
        server_pollfd.revents = 0;
        fds.push_back(server_pollfd);
        
        std::cout &lt;&lt; "服务器初始化成功，监听端口: " &lt;&lt; port &lt;&lt; std::endl;
        return true;
    }
    
    void run() {
        std::cout &lt;&lt; "服务器开始运行..." &lt;&lt; std::endl;
        
        while (true) {
            // 调用poll等待事件
            int ret = poll(fds.data(), fds.size(), -1);  // -1表示无限等待
            
            if (ret &lt; 0) {
                std::cerr &lt;&lt; "poll错误: " &lt;&lt; strerror(errno) &lt;&lt; std::endl;
                break;
            }
            
            if (ret == 0) {
                // 超时，但这里不会发生，因为timeout=-1
                continue;
            }
            
            // 检查所有文件描述符
            for (size_t i = 0; i &lt; fds.size(); i++) {
                if (fds[i].revents == 0) {
                    continue;
                }
                
                // 处理错误或挂断事件
                if (fds[i].revents &amp; (POLLERR | POLLHUP | POLLNVAL)) {
                    if (fds[i].fd == server_fd) {
                        std::cerr &lt;&lt; "服务器socket错误，停止服务" &lt;&lt; std::endl;
                        return;
                    } else {
                        std::cout &lt;&lt; "客户端连接关闭或出错: fd=" &lt;&lt; fds[i].fd &lt;&lt; std::endl;
                        close(fds[i].fd);
                        fds.erase(fds.begin() + i);
                        i--;  // 调整索引
                        continue;
                    }
                }
                
                // 处理可读事件
                if (fds[i].revents &amp; POLLIN) {
                    if (fds[i].fd == server_fd) {
                        // 有新的客户端连接
                        handleNewConnection();
                    } else {
                        // 客户端有数据可读
                        handleClientData(i);
                    }
                }
            }
        }
    }
    
private:
    void handleNewConnection() {
        struct sockaddr_in client_addr;
        socklen_t addr_len = sizeof(client_addr);
        
        // 接受新连接
        int client_fd = accept(server_fd, (struct sockaddr*)&amp;client_addr, &amp;addr_len);
        
        if (client_fd &lt; 0) {
            std::cerr &lt;&lt; "接受连接失败: " &lt;&lt; strerror(errno) &lt;&lt; std::endl;
            return;
        }
        
        // 设置非阻塞模式
        fcntl(client_fd, F_SETFL, fcntl(client_fd, F_GETFL, 0) | O_NONBLOCK);
        
        // 添加新的客户端到pollfd数组
        pollfd client_pollfd;
        client_pollfd.fd = client_fd;
        client_pollfd.events = POLLIN;
        client_pollfd.revents = 0;
        fds.push_back(client_pollfd);
        
        std::cout &lt;&lt; "新客户端连接: fd=" &lt;&lt; client_fd 
                  &lt;&lt; " IP=" &lt;&lt; inet_ntoa(client_addr.sin_addr) 
                  &lt;&lt; ":" &lt;&lt; ntohs(client_addr.sin_port) &lt;&lt; std::endl;
    }
    
    void handleClientData(size_t index) {
        char buffer[1024];
        ssize_t bytes_read;
        
        // 读取数据
        bytes_read = recv(fds[index].fd, buffer, sizeof(buffer) - 1, 0);
        
        if (bytes_read &lt;= 0) {
            // 连接关闭或错误
            if (bytes_read == 0) {
                std::cout &lt;&lt; "客户端断开连接: fd=" &lt;&lt; fds[index].fd &lt;&lt; std::endl;
            } else {
                std::cerr &lt;&lt; "读取错误: " &lt;&lt; strerror(errno) &lt;&lt; std::endl;
            }
            close(fds[index].fd);
            fds.erase(fds.begin() + index);
            return;
        }
        
        // 确保字符串以null结尾
        buffer[bytes_read] = '\0';
        
        std::cout &lt;&lt; "收到客户端 " &lt;&lt; fds[index].fd &lt;&lt; " 的数据: " &lt;&lt; buffer;
        
        // 回显数据给客户端
        std::string response = "服务器收到: ";
        response += buffer;
        
        ssize_t bytes_sent = send(fds[index].fd, response.c_str(), response.length(), 0);
        
        if (bytes_sent &lt; 0) {
            std::cerr &lt;&lt; "发送失败: " &lt;&lt; strerror(errno) &lt;&lt; std::endl;
            close(fds[index].fd);
            fds.erase(fds.begin() + index);
        }
    }
    
    void cleanup() {
        // 关闭所有客户端连接
        for (size_t i = 0; i &lt; fds.size(); i++) {
            if (fds[i].fd &gt;= 0) {
                close(fds[i].fd);
            }
        }
        fds.clear();
        
        // 关闭服务器socket
        if (server_fd &gt;= 0) {
            close(server_fd);
            server_fd = -1;
        }
        
        std::cout &lt;&lt; "服务器资源已清理" &lt;&lt; std::endl;
    }
};

int main() {
    PollServer server(8888);
    
    if (!server.initialize()) {
        std::cerr &lt;&lt; "服务器初始化失败" &lt;&lt; std::endl;
        return 1;
    }
    
    server.run();
    
    return 0;
}</code></pre><p style=""></p>
